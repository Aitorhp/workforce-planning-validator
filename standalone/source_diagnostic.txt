  function classifyShiftPeriod(shiftStartMs) { return dtHour(shiftStartMs) < 13 ? "MAÑANA" : "TARDE";
 }

  function analyzeShiftBalance(shifts) {
    const grouped = new Map();

    for (const s of shifts) {
      const k = s.store + "\u0001" + s.person;

      if (!grouped.has(k)) grouped.set(k, { store: s.store, person: s.person, tm: 0, tt: 0, hm: 0, ht: 0 }
);

      const g = grouped.get(k);

      if (classifyShiftPeriod(s.shiftStart) === "MAÑANA") { g.tm++;
 g.hm += s.workedHours;
 }

      else { g.tt++;
 g.ht += s.workedHours;
 }

    }

    const rows = [...grouped.values()].sort((a, b) => String(a.store) < String(b.store) ? -1 : String(a.store) > String(b.store) ? 1 :
      String(a.person) < String(b.person) ? -1 : String(a.person) > String(b.person) ? 1 : 0);

    return rows.map(g => {
      const total = g.tm + g.tt;

      const mp = total ? g.tm / total * 100 : 0, ap = total ? g.tt / total * 100 : 0;

      const bal = total ? 2 * Math.min(g.tm, g.tt) / total * 100 : 0;

      const rot = (g.tm && g.tt) ? "Mañana y tarde" : g.tm ? "Solo mañanas" : "Solo tardes";

      return {
        id_tienda: g.store, personId: g.person, turnos_manana: g.tm, turnos_tarde: g.tt, turnos_totales: total,
        horas_manana: pyround(g.hm, 4), horas_tarde: pyround(g.ht, 4),
        porcentaje_manana: pyround(mp, 2), porcentaje_tarde: pyround(ap, 2),
        sesgo_tarde_pct: pyround(ap - mp, 2), indice_equilibrio_pct: pyround(bal, 2),
        rota_manana_tarde: (g.tm && g.tt) ? "SI" : "NO", estado_rotacion: rot,
      }
;

    }
);

  }

  function analyzeDailyAbsences(absences, dataDates) {
    const byDate = new Map();

    for (const a of absences) { if (!byDate.has(a.day)) byDate.set(a.day, []);
 byDate.get(a.day).push(a);
 }

    const dates = (dataDates && dataDates.size ? [...dataDates] : [...byDate.keys()]).sort((a, b) => a - b);

    return dates.map(d => {
      const daily = byDate.get(d) || [];

      const emp = new Set(daily.map(a => a.store + "\u0001" + a.person));

      const types = [...new Set(daily.map(a => a.type))].sort();

      return { fecha: dayISO(d), fecha_day: d, empleados_ausentes: emp.size, registros_ausencia: daily.length, tipos_ausencia: types.join(", ") }
;

    }
);

  }

  function extractData(data, scheduleSource) {
    const storeId = (data.store || {}
).id;

    const storeDayTimes = data.storeDayTimes || [];

    if (!Array.isArray(storeDayTimes)) throw new Error("El campo 'storeDayTimes' debe ser una lista.");

    const shifts = [];

    const employeeMonths = new Map();
          // "store|person|YYYY-MM" -> applicable
    const employeeMonthsMeta = new Map();
      // key -> {store, person, month}

    const absences = [];

    const presence = new Map();
                // "store|person" -> Set(dayInt)
    const presenceMeta = new Map();

    const maxBreakH = CONFIG.calculation.max_internal_break_hours;


    for (const storeDay of storeDayTimes) {
      if (!storeDay || typeof storeDay !== "object" || !storeDay.operatingDate) continue;

      const operatingDay = parseDay(storeDay.operatingDate);

      for (const personDay of storeDay.people || []) {
        if (!personDay || typeof personDay !== "object") continue;

        const person = personDay.person || {}
;

        const personId = personDay.personId != null ? personDay.personId : person.personId;

        const applicable = person.applicableWorkingHours;

        const empKey = storeId + "\u0001" + personId;

        if (!presence.has(empKey)) { presence.set(empKey, new Set());
 presenceMeta.set(empKey, { store: storeId, person: personId }
);
 }

        presence.get(empKey).add(operatingDay);

        const mKey = empKey + "\u0001" + monthKey(operatingDay);

        employeeMonths.set(mKey, applicable);

        employeeMonthsMeta.set(mKey, { store: storeId, person: personId, month: monthKey(operatingDay) }
);


        const dayTimes = personDay.dayTimes || {}
;

        // Ausencias
        const seen = new Set();

        for (const absence of dayTimes.absences || []) {
          if (!absence || typeof absence !== "object") continue;

          const status = String(absence.status || "").toUpperCase();

          if (status !== "VALIDATED" && status !== "APPROVED") continue;

          const t = absence.type || {}
;

          const absType = String(t.name || t.description || absence.id || "AUSENCIA");

          const k = absType + "\u0001" + status;

          if (seen.has(k)) continue;

          seen.add(k);

          absences.push({ store: storeId, person: personId, day: operatingDay, type: absType, status }
);

        }

        // Segmentos WORK
        let selected = dayTimes[scheduleSource];

        if (!Array.isArray(selected)) selected = [];

        const segs = [];

        for (const seg of selected) {
          if (!isWork(seg)) continue;

          const sv = seg.startDateTime, ev = seg.endDateTime;

          if (!sv || !ev) continue;

          const start = parseDateTime(sv), end = parseDateTime(ev);

          if (end <= start) throw new Error(`Segmento invalido en ${scheduleSource}
: personId=${personId}
`);

          segs.push([start, end]);

        }

        if (!segs.length) continue;

        segs.sort((a, b) => a[0] - b[0]);

        const shiftStart = segs[0][0];

        const shiftEnd = Math.max(...segs.map(s => s[1]));

        let netMs = segs.reduce((a, s) => a + (s[1] - s[0]), 0);

        let breakMs = 0, prevEnd = segs[0][1];

        for (let i = 1;
 i < segs.length;
 i++) {
          const [cs, ce] = segs[i];

          const gap = cs - prevEnd;

          if (gap > 0 && gap <= maxBreakH * 3600000) breakMs += gap;

          if (ce > prevEnd) prevEnd = ce;

        }

        shifts.push({
          store: storeId, person: personId, applicable, workDay: operatingDay,
          shiftStart, shiftEnd,
          workedHours: pyround(netMs / 3600000, 4),
          breakHours: pyround(breakMs / 3600000, 4),
        }
);

      }

    }

    const scmp = (a, b) => (String(a.store) < String(b.store) ? -1 : String(a.store) > String(b.store) ? 1 :
      String(a.person) < String(b.person) ? -1 : String(a.person) > String(b.person) ? 1 : 0);

    shifts.sort((a, b) => scmp(a, b) || a.workDay - b.workDay);

    absences.sort((a, b) => scmp(a, b) || a.day - b.day);

    return { shifts, employeeMonths, employeeMonthsMeta, absences, presence, presenceMeta, storeId }
;

  }

  function findConsecutiveStreaks(days) {
    if (!days.length) return [];

    const uniq = [...new Set(days)].sort((a, b) => a - b);

    const streaks = [[uniq[0]]];

    for (let i = 1;
 i < uniq.length;
 i++) {
      if (uniq[i] === streaks[streaks.length - 1][streaks[streaks.length - 1].length - 1] + 1)
        streaks[streaks.length - 1].push(uniq[i]);

      else streaks.push([uniq[i]]);

    }

    return streaks;

  }

function parsePlanningText(text, filename){
  let data;

  try{ data=JSON.parse(String(text).replace(/^﻿/,""));
 }

  catch(e){ throw new Error(t("JSON no valido: {e}
",{e:`${filename}
: ${e.message}
`}
));
 }

  if(!data || typeof data!=="object" || Array.isArray(data)) throw new Error(t("La raiz del JSON debe ser un objeto."));

  return data;

}

function planningDocumentInfo(data, filename){
  const storeId=(data.store||{}
).id;

  if(storeId===null || storeId===undefined || storeId==="") throw new Error(`${filename}
: ${t("El fichero no identifica la tienda.")}
`);

  if(!Array.isArray(data.storeDayTimes)) throw new Error(`${filename}
: ${t("El campo storeDayTimes debe ser una lista.")}
`);

  const dates=[];
 const seen=new Set();

  for(const item of data.storeDayTimes){
    if(!item || typeof item!=="object" || !item.operatingDate) continue;

    const day=D.parseDay(item.operatingDate);

    if(seen.has(day)) throw new Error(`${filename}
: ${t("Los periodos de los ficheros no pueden solaparse.")}
`);

    seen.add(day);
 dates.push(day);

  }

  dates.sort((a,b)=>a-b);

  if(!dates.length) throw new Error(`${filename}
: ${t("No se han encontrado fechas operativas en el fichero.")}
`);

  const months=uniq(dates.map(D.monthKey));

  if(months.length!==1) throw new Error(`${filename}
: ${t("Cada fichero debe contener un único mes calendario.")}
`);

  return {data,filename,storeId,dates,dateSet:new Set(dates),month:months[0],first:dates[0],last:dates[dates.length-1]}
;

}

function monthOrdinal(month){ const [y,m]=month.split("-").map(Number);
 return y*12+(m-1);
 }

function combinePlanningDocuments(entries){
  if(!entries.length) throw new Error(t("Sube uno o dos ficheros para detectar los orígenes de horarios disponibles."));

  if(entries.length>2) throw new Error(t("Solo se admite un máximo de dos ficheros."));

  const infos=entries.map(e=>planningDocumentInfo(e.data,e.filename)).sort((a,b)=>a.first-b.first);

  if(infos.length===2){
    if(String(infos[0].storeId)!==String(infos[1].storeId)) throw new Error(t("Los dos ficheros deben pertenecer a la misma tienda."));

    if(Math.abs(monthOrdinal(infos[1].month)-monthOrdinal(infos[0].month))!==1) throw new Error(t("Los meses deben ser consecutivos."));

    for(const day of infos[0].dateSet) if(infos[1].dateSet.has(day)) throw new Error(t("Los periodos de los ficheros no pueden solaparse."));

  }

  const combined=JSON.parse(JSON.stringify(infos[0].data));

  combined.storeDayTimes=infos.flatMap(info=>info.data.storeDayTimes || []).slice().sort((a,b)=>{ const ad=a&&a.operatingDate?D.parseDay(a.operatingDate):Number.MAX_SAFE_INTEGER;
 const bd=b&&b.operatingDate?D.parseDay(b.operatingDate):Number.MAX_SAFE_INTEGER;
 return ad-bd;
 }
);

  return {data:combined,infos,periodStart:infos[0].first,periodEnd:infos[infos.length-1].last}
;

}

function activePlanningWeekSet(result){
  const set=new Set();

  (result.shifts||[]).forEach(s=>{ if(Number(s.horas_totales)>0) set.add(D.weekStart(s.day_int));
 }
);

  return set;

}

function computeWeekStarts(result){ return [...activePlanningWeekSet(result)].sort((a,b)=>a-b);
 }

function buildFrames(){
  const r=S.result, activeWeeks=activePlanningWeekSet(r);

  const activeDay=d=>activeWeeks.has(D.weekStart(d));

  const activeDates=r.data_dates.filter(activeDay);

  const full = {
    shifts:r.shifts.filter(s=>activeDay(s.day_int)),
    weekly:[],
    absences:r.absences.filter(a=>activeDay(a.fecha_int)),
    absence_daily:r.absence_daily.filter(a=>activeDay(a.fecha_day)),
    incidents:r.incidents,
    summaries:r.summaries,
    data_dates:activeDates,
  }
;

  // Weekly rows use ISO strings;
 avoid relying on a non-exported helper.
  full.weekly=r.weekly.filter(w=>activeWeeks.has(D.weekStart(di(w.inicio_semana))));

  if(S.weekIdx<0){ full.dataSet=new Set(full.data_dates);
 return full;
 }

  const ws=S.weekStarts[S.weekIdx], we=ws+6;

  const inRange=d=> d>=ws && d<=we;

  const wsISO=D.dayISO(ws);

  const dd=activeDates.filter(inRange);

  return {
    shifts:full.shifts.filter(s=>inRange(s.day_int)),
    weekly:full.weekly.filter(w=>w.inicio_semana===wsISO),
    absences:full.absences.filter(a=>inRange(a.fecha_int)),
    absence_daily:full.absence_daily.filter(a=>inRange(a.fecha_day)),
    incidents:full.incidents.filter(i=> i.fecha_inicio_int<=we && i.fecha_fin_int>=ws),
    summaries:r.summaries,
    data_dates:dd, dataSet:new Set(dd),
  }
;

}

function selectSource(src){
  S.source=src;
 S.tab=0;
 S.weekIdx=-1;

  S.f={ wkDev:true, wkStatus:null, wkWeek:"Todas", wkEmp:"", balNon:true, morningCutoff:"11:00", afternoonCutoff:"14:00", wkendMin:null, wkendMax:null, wkendReqFull:1, wkendReqSat:1, wkendReqSun:1, wkendAlerts:true, wkendDiag:0, wkendMapQuery:"", wkendMapAlerts:false }
;

  if(!S.cache[src]) S.cache[src]=D.runValidation(S.raw,src);

  S.result=S.cache[src];

  S.weekStarts=computeWeekStarts(S.result);

  S.frames=buildFrames();

  renderSidebarDynamic();

  renderHeader();

  renderContent();

}

function setWeek(idx){ S.weekIdx=idx;
 S.frames=buildFrames();

  S.f.wkendMin=null;
 S.f.wkendMax=null;
 S.f.wkStatus=null;

  renderHeader();
 renderContent();
 }

  document.getElementById("introMsg").style.display="none";

  const fileMessage=S.filenames.length>1?t("Se han combinado {n}

  const html=`<div class="source-box"><b>${t("Origen analizado:")}

)}

function downloadExcel(){
  const r=S.frames;
 const src=S.source;

  const sheets=[];

  sheets.push({name:t("Informacion"),rows:[[t("Origen analizado"),src],[t("Descripcion"),t(D.SCHEDULE_SOURCES[src])],[t("Ficheros"),S.filenames.join(" | ")],[t("Periodo combinado"),S.combinedPeriod?D.dayISO(S.combinedPeriod.start)+" - "+D.dayISO(S.combinedPeriod.end):""],[t("Los turnos de mañana empiezan antes de"),S.f.morningCutoff],[t("Los turnos de tarde empiezan después de"),S.f.afternoonCutoff],[t("Mínimo fines de semana completos libres"),S.f.wkendReqFull],[t("Mínimo sábados libres"),S.f.wkendReqSat],[t("Mínimo domingos libres"),S.f.wkendReqSun]]}
);

  const sh=r.shifts.map(s=>[s.id_tienda,s.personId,s.applicableWorkingHours,s.day,s.hora_inicio,s.hora_fin,s.horas_totales,s.duracion_descanso,configurableShiftPeriod(s.hora_inicio,S.f.morningCutoff,S.f.afternoonCutoff)]);

  sheets.push({name:t("Turnos"),rows:[["id_tienda","personId","applicableWorkingHours","day","hora_inicio","hora_fin","horas_totales","duracion_descanso","franja_turno"],...sh]}
);

  const sm=r.summaries;
 const smc=sm.length?Object.keys(sm[0]):[t("Sin registros")];

  sheets.push({name:t("Validacion mensual"),rows:[smc,...sm.map(o=>smc.map(k=>o[k]))]}
);

  const cc=contractChanges(sm);

  sheets.push({name:t("Cambios contractuales"),rows:[["id_tienda","personId","Mes anterior","Horas mes anterior","Mes siguiente","Horas mes siguiente","Diferencia","Estado"],...cc.map(o=>[o.id_tienda,o.personId,o["Mes anterior"],o["Horas mes anterior"],o["Mes siguiente"],o["Horas mes siguiente"],o["Diferencia"],o.Estado])]}
);

  const inc=r.incidents.map(i=>[i.id_tienda,i.personId,i.mes,i.tipo_incidencia,i.fecha_inicio,i.fecha_fin,i.valor_observado,i.limite,i.detalle]);

  sheets.push({name:t("Detalle incidencias"),rows:[["id_tienda","personId","mes","tipo_incidencia","fecha_inicio","fecha_fin","valor_observado","limite","detalle"],...inc]}
);

  const wk=r.weekly;
 const wkc=wk.length?Object.keys(wk[0]):[t("Sin registros")];

  sheets.push({name:t("Control horas semanal"),rows:[wkc,...wk.map(o=>wkc.map(k=>o[k]))]}
);

  const ab=r.absences.map(a=>[a.id_tienda,a.personId,a.fecha,a.tipo_ausencia,a.estado]);

  sheets.push({name:t("Ausencias"),rows:[["id_tienda","personId","fecha","tipo_ausencia","estado"],...ab]}
);

  const bytes=buildXlsx(sheets);

  const blob=new Blob([bytes],{type:"application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"}
);

  const a=document.createElement("a");
 a.href=URL.createObjectURL(blob);
 a.download=`validacion_${src}
.xlsx`;
 a.click();

  setTimeout(()=>URL.revokeObjectURL(a.href),2000);

}
