  function analyzeShiftBalance(shifts) {
    const grouped = new Map();

    for (const s of shifts) {
      const k = s.store + "\u0001" + s.person;

      if (!grouped.has(k)) grouped.set(k, { store: s.store, person: s.person, tm: 0, tt: 0, hm: 0, ht: 0 }
);

      const g = grouped.get(k);

      if (classifyShiftPeriod(s.shiftStart) === "MAÑANA") { g.tm++;
 g.hm += s.workedHours;
 }

      else { g.tt++;
 g.ht += s.workedHours;
 }

    }

    const rows = [...grouped.values()].sort((a, b) => String(a.store) < String(b.store) ? -1 : String(a.store) > String(b.store) ? 1 :
      String(a.person) < String(b.person) ? -1 : String(a.person) > String(b.person) ? 1 : 0);

    return rows.map(g => {
      const total = g.tm + g.tt;

      const mp = total ? g.tm / total * 100 : 0, ap = total ? g.tt / total * 100 : 0;

      const bal = total ? 2 * Math.min(g.tm, g.tt) / total * 100 : 0;

      const rot = (g.tm && g.tt) ? "Mañana y tarde" : g.tm ? "Solo mañanas" : "Solo tardes";

      return {
        id_tienda: g.store, personId: g.person, turnos_manana: g.tm, turnos_tarde: g.tt, turnos_totales: total,
        horas_manana: pyround(g.hm, 4), horas_tarde: pyround(g.ht, 4),
        porcentaje_manana: pyround(mp, 2), porcentaje_tarde: pyround(ap, 2),
        sesgo_tarde_pct: pyround(ap - mp, 2), indice_equilibrio_pct: pyround(bal, 2),
        rota_manana_tarde: (g.tm && g.tt) ? "SI" : "NO", estado_rotacion: rot,
      }
;

    }
);

  }

function prepWeeklyDetail(weekly){
  const ev=weekly.filter(w=>["COINCIDE","FALTAN HORAS","EXCESO HORAS"].includes(w.estado_planificacion));

  const nonEv=weekly.filter(w=>!["COINCIDE","FALTAN HORAS","EXCESO HORAS"].includes(w.estado_planificacion));

  const labels={"PODRIA EXPLICAR TODAS LAS HORAS FALTANTES":t("Sí, explica todo el déficit"),"PODRIA EXPLICAR PARTE DE LAS HORAS FALTANTES":t("Explica parte del déficit"),"AUSENCIA SIN MEDIA CALCULABLE":t("Sin estimación disponible"),"AUSENTE TODO EL PERIODO":t("Ausente todo el periodo"),"NO":t("No explicado por ausencias")}
;

  const rows=ev.map(w=>{
    const c=(x)=> x==null?0:x;

    const app=c(w.applicableWorkingHours), plan=c(w.horas_planificadas), miss=c(w.horas_no_planificadas_hasta_contrato), pot=c(w.horas_potenciales_asociadas_ausencia);

    const apl=Math.min(miss,pot);

    let expl = labels[w.posible_explicacion_por_ausencia]||t("No explicado por ausencias");

    if(w.estado_planificacion!=="FALTAN HORAS") expl=t("Contrato cubierto o con exceso");

    let ctx=t("No aplica");

    if(w.estado_planificacion==="FALTAN HORAS") ctx = w.dias_ausencia_sin_turno>0?t("Con ausencia registrada"):t("Sin ausencia registrada");

    if(w.ausente_todo_el_periodo==="SI") ctx=t("Ausente todo el periodo");

    return {...w, Semana:weekLabel(w.ano_iso,w.semana_iso), Empleado:`${w.id_tienda}
 · ${w.personId}
`,
      _app:app,_plan:plan,_miss:miss,_exc:c(w.horas_planificadas_en_exceso),
      desviacion:plan-app, cobertura: app!==0?plan/app*100:null,
      horas_ausencia_aplicables:apl, planificadas_mas_ausencia:plan+apl,
      deficit_explicable: w.posible_explicacion_por_ausencia==="PODRIA EXPLICAR TODAS LAS HORAS FALTANTES",
      explicacion:expl, contexto:ctx }
;

  }
);

  return {rows,nonEv}
;

}

function prepWeeklyTotals(rows){
  const g=groupBy(rows,r=>r.ano_iso+"|"+r.semana_iso);

  const out=[...g.values()].map(rs=>({ano_iso:rs[0].ano_iso,semana_iso:rs[0].semana_iso,inicio:di(rs[0].inicio_semana),Semana:rs[0].Semana,
    horas_contrato:sum(rs.map(r=>r._app)),horas_planificadas:sum(rs.map(r=>r._plan)),
    horas_faltantes:sum(rs.map(r=>r._miss)),horas_exceso:sum(rs.map(r=>r._exc)),
    horas_ausencia_estimadas:sum(rs.map(r=>Number(r.horas_potenciales_asociadas_ausencia)||0))}
));

  out.sort((a,b)=>a.ano_iso-b.ano_iso||a.semana_iso-b.semana_iso);

  return out;

}

  function findConsecutiveStreaks(days) {
    if (!days.length) return [];

    const uniq = [...new Set(days)].sort((a, b) => a - b);

    const streaks = [[uniq[0]]];

    for (let i = 1;
 i < uniq.length;
 i++) {
      if (uniq[i] === streaks[streaks.length - 1][streaks[streaks.length - 1].length - 1] + 1)
        streaks[streaks.length - 1].push(uniq[i]);

      else streaks.push([uniq[i]]);

    }

    return streaks;

  }

    return rows;

  }


  // ---- Analítica: balance y ausencias diarias ------------------------------
  function classifyShiftPeriod(shiftStartMs) { return dtHour(shiftStartMs) < 13 ? "MAÑANA" : "TARDE";
 }

  function analyzeShiftBalance(shifts) {
    const grouped = new Map();

    for (const s of shifts) {
      const k = s.store + "\u0001" + s.person;

      if (!grouped.has(k)) grouped.set(k, { store: s.store, person: s.person, tm: 0, tt: 0, hm: 0, ht: 0 }
);

      const g = grouped.get(k);

      if (classifyShiftPeriod(s.shiftStart) === "MAÑANA") { g.tm++;
 g.hm += s.workedHours;
 }

      else { g.tt++;
 g.ht += s.workedHours;
 }

    }

    const rows = [...grouped.values()].sort((a, b) => String(a.store) < String(b.store) ? -1 : String(a.store) > String(b.store) ? 1 :
      String(a.person) < String(b.person) ? -1 : String(a.person) > String(b.person) ? 1 : 0);

    return rows.map(g => {
      const total = g.tm + g.tt;

      const mp = total ? g.tm / total * 100 : 0, ap = total ? g.tt / total * 100 : 0;

      const bal = total ? 2 * Math.min(g.tm, g.tt) / total * 100 : 0;

      const rot = (g.tm && g.tt) ? "Mañana y tarde" : g.tm ? "Solo mañanas" : "Solo tardes";

      return {
        id_tienda: g.store, personId: g.person, turnos_manana: g.tm, turnos_tarde: g.tt, turnos_totales: total,
        horas_manana: pyround(g.hm, 4), horas_tarde: pyround(g.ht, 4),
        porcentaje_manana: pyround(mp, 2), porcentaje_tarde: pyround(ap, 2),
        sesgo_tarde_pct: pyround(ap - mp, 2), indice_equilibrio_pct: pyround(bal, 2),
        rota_manana_tarde: (g.tm && g.tt) ? "SI" : "NO", estado_rotacion: rot,
      }
;

    }
);

  }

  function analyzeDailyAbsences(absences, dataDates) {
    const byDate = new Map();

    for (const a of absences) { if (!byDate.has(a.day)) byDate.set(a.day, []);
 byDate.get(a.day).push(a);
 }

    const dates = (dataDates && dataDates.size ? [...dataDates] : [...byDate.keys()]).sort((a, b) => a - b);

    return dates.map(d => {
      const daily = byDate.get(d) || [];

      const emp = new Set(daily.map(a => a.store + "\u0001" + a.person));

      const types = [...new Set(daily.map(a => a.type))].sort();

      return { fecha: dayISO(d), fecha_day: d, empleados_ausentes: emp.size, registros_ausencia: daily.length, tipos_ausencia: types.join(", ") }
;

    }
);

  }
