function prepWeekend(F){
  const dates=[...F.dataSet].sort((a,b)=>a-b);
 const dateSet=F.dataSet;

  // empleados: applicable máximo (o null si algún null)
  const g=groupBy(F.weekly,w=>w.id_tienda+"|"+w.personId);

  const employees=[...g.values()].map(rs=>{ const vals=rs.map(r=>r.applicableWorkingHours);

    const app = vals.some(v=>v==null)?null:Math.max(...vals.map(Number));

    return {id_tienda:rs[0].id_tienda,personId:rs[0].personId,app}
;
 }
);

  const worked=new Set(F.shifts.map(s=>s.id_tienda+"|"+s.personId+"|"+s.day_int));

  const months=uniq(dates.map(d=>{const {y,m}
=D.dayYMD(d);
return y+"-"+m;
}
));

  const monthly=[],detail=[];

  employees.forEach(emp=>{
    months.forEach(ym=>{ const [Y,M]=ym.split("-").map(Number);

      const md=dates.filter(d=>{const {y,m}
=D.dayYMD(d);
return y===Y&&m===M;
}
);

      const sats=md.filter(d=>D.weekdayISO(d)===5), suns=md.filter(d=>D.weekdayISO(d)===6);

      const pairs=sats.filter(s=>dateSet.has(s+1)).map(s=>[s,s+1]);

      const freeSat=sats.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      const freeSun=suns.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      let freeWk=0;

      pairs.forEach(([sat,sun])=>{ const sf=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sat), su=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sun);

        if(sf&&su)freeWk++;

        detail.push({id_tienda:emp.id_tienda,personId:emp.personId,ini:sat,fin:`${ddmm(sat)}
 - ${ddmm(sun)}
`,sat:sf,sun:su,libre:sf&&su}
);
 }
);

      monthly.push({id_tienda:emp.id_tienda,personId:emp.personId,app:emp.app,Mes:`${Y}
-${String(M).padStart(2,"0")}
`,
        fs_eval:pairs.length,fs_libres:freeWk,sab_eval:sats.length,sab_libres:freeSat,dom_eval:suns.length,dom_libres:freeSun}
);
 }
);

  }
);

  return {monthly,detail}
;

}
