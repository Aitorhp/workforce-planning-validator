  function findConsecutiveStreaks(days) {
    if (!days.length) return [];

    const uniq = [...new Set(days)].sort((a, b) => a - b);

    const streaks = [[uniq[0]]];

    for (let i = 1;
 i < uniq.length;
 i++) {
      if (uniq[i] === streaks[streaks.length - 1][streaks[streaks.length - 1].length - 1] + 1)
        streaks[streaks.length - 1].push(uniq[i]);

      else streaks.push([uniq[i]]);

    }

    return streaks;

  }

  function weekendCounts(year, month, workedSet) {
    const daysInMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();

    let freeSat = 0, freeSun = 0, complete = 0;

    for (let d = 1;
 d <= daysInMonth;
 d++) {
      const day = dayFromYMD(year, month, d);

      const wd = weekdayISO(day);

      if (wd === 5) { // sábado
        if (!workedSet.has(day)) freeSat++;

        const nxt = day + 1;

        if (!workedSet.has(day) && dayYMD(nxt).m === month && !workedSet.has(nxt)) complete++;

      }
 else if (wd === 6) { // domingo
        if (!workedSet.has(day)) freeSun++;

      }

    }

    return [complete, freeSat, freeSun];

  }

    central.push([0x50,0x4b,1,2,...u16(20),...u16(20),...u16(0),...u16(0),...u16(0),...u16(0),...u32(crc),...u32(data.length),...u32(data.length),...u16(f.name.length),...u16(0),...u16(0),...u16(0),...u16(0),...u32(0),...u32(offset),...strBytes(f.name)]);

    offset+=local.length+data.length;

  }

  const cd=[];
 central.forEach(c=>cd.push(...c));
 const cdBytes=new Uint8Array(cd);

  const end=[0x50,0x4b,5,6,...u16(0),...u16(0),...u16(central.length),...u16(central.length),...u32(cdBytes.length),...u32(offset),...u16(0)];

  const out=new Uint8Array(chunks.reduce((a,c)=>a+c.length,0)+cdBytes.length+end.length);

  let p=0;
 for(const c of chunks){ out.set(c,p);
 p+=c.length;
 }
 out.set(cdBytes,p);
 p+=cdBytes.length;
 out.set(new Uint8Array(end),p);

  return out;

}

function xlsxCell(v,ref){
  if(v==null||v==="") return `<c r="${ref}
"/>`;

  if(typeof v==="number" && isFinite(v)) return `<c r="${ref}
"><v>${v}
</v></c>`;

  const s=String(v).replace(/&/g,"&amp;
").replace(/</g,"&lt;
").replace(/>/g,"&gt;
");

  return `<c r="${ref}
" t="inlineStr"><is><t xml:space="preserve">${s}
</t></is></c>`;

}

function parsePlanningText(text, filename){
  let data;

  try{ data=JSON.parse(String(text).replace(/^﻿/,""));
 }

  catch(e){ throw new Error(t("JSON no valido: {e}
",{e:`${filename}
: ${e.message}
`}
));
 }

  if(!data || typeof data!=="object" || Array.isArray(data)) throw new Error(t("La raiz del JSON debe ser un objeto."));

function planningDocumentInfo(data, filename){
  const storeId=(data.store||{}
).id;

  if(storeId===null || storeId===undefined || storeId==="") throw new Error(`${filename}
: ${t("El fichero no identifica la tienda.")}
`);

  if(!Array.isArray(data.storeDayTimes)) throw new Error(`${filename}
: ${t("El campo storeDayTimes debe ser una lista.")}
`);

  const dates=[];
 const seen=new Set();

  for(const item of data.storeDayTimes){
    if(!item || typeof item!=="object" || !item.operatingDate) continue;

    const day=D.parseDay(item.operatingDate);

    if(seen.has(day)) throw new Error(`${filename}
: ${t("Los periodos de los ficheros no pueden solaparse.")}
`);

    seen.add(day);
 dates.push(day);

  }

  dates.sort((a,b)=>a-b);

  if(!dates.length) throw new Error(`${filename}
: ${t("No se han encontrado fechas operativas en el fichero.")}
`);

  const months=uniq(dates.map(D.monthKey));

  if(months.length!==1) throw new Error(`${filename}
: ${t("Cada fichero debe contener un único mes calendario.")}
`);

  return {data,filename,storeId,dates,dateSet:new Set(dates),month:months[0],first:dates[0],last:dates[dates.length-1]}
;

}

function monthOrdinal(month){ const [y,m]=month.split("-").map(Number);
 return y*12+(m-1);
 }

function combinePlanningDocuments(entries){
  if(!entries.length) throw new Error(t("Sube uno o dos ficheros para detectar los orígenes de horarios disponibles."));

  if(entries.length>2) throw new Error(t("Solo se admite un máximo de dos ficheros."));

  const infos=entries.map(e=>planningDocumentInfo(e.data,e.filename)).sort((a,b)=>a.first-b.first);

  if(infos.length===2){
    if(String(infos[0].storeId)!==String(infos[1].storeId)) throw new Error(t("Los dos ficheros deben pertenecer a la misma tienda."));

    if(Math.abs(monthOrdinal(infos[1].month)-monthOrdinal(infos[0].month))!==1) throw new Error(t("Los meses deben ser consecutivos."));

    for(const day of infos[0].dateSet) if(infos[1].dateSet.has(day)) throw new Error(t("Los periodos de los ficheros no pueden solaparse."));

  }

  const combined=JSON.parse(JSON.stringify(infos[0].data));

  combined.storeDayTimes=infos.flatMap(info=>info.data.storeDayTimes || []).slice().sort((a,b)=>{ const ad=a&&a.operatingDate?D.parseDay(a.operatingDate):Number.MAX_SAFE_INTEGER;
 const bd=b&&b.operatingDate?D.parseDay(b.operatingDate):Number.MAX_SAFE_INTEGER;
 return ad-bd;
 }
);

  return {data:combined,infos,periodStart:infos[0].first,periodEnd:infos[infos.length-1].last}
;

}

function buildFrames(){
  const r=S.result, activeWeeks=activePlanningWeekSet(r);

  const activeDay=d=>activeWeeks.has(D.weekStart(d));

  const activeDates=r.data_dates.filter(activeDay);

  const full = {
    shifts:r.shifts.filter(s=>activeDay(s.day_int)),
    weekly:[],
    absences:r.absences.filter(a=>activeDay(a.fecha_int)),

    incidents:r.incidents,
    summaries:r.summaries,
    data_dates:activeDates,
  }
;

  // Weekly rows use ISO strings;
 avoid relying on a non-exported helper.
  full.weekly=r.weekly.filter(w=>activeWeeks.has(D.weekStart(di(w.inicio_semana))));

  if(S.weekIdx<0){ full.dataSet=new Set(full.data_dates);
 return full;
 }

  const ws=S.weekStarts[S.weekIdx], we=ws+6;

  const inRange=d=> d>=ws && d<=we;

    weekly:full.weekly.filter(w=>w.inicio_semana===wsISO),

  }
;

}

/* ---------- Fines de semana ---------- */
function prepWeekend(F){
  const dates=[...F.dataSet].sort((a,b)=>a-b);
 const dateSet=F.dataSet;

  // empleados: applicable máximo (o null si algún null)
  const g=groupBy(F.weekly,w=>w.id_tienda+"|"+w.personId);

  const employees=[...g.values()].map(rs=>{ const vals=rs.map(r=>r.applicableWorkingHours);

    const app = vals.some(v=>v==null)?null:Math.max(...vals.map(Number));

    return {id_tienda:rs[0].id_tienda,personId:rs[0].personId,app}
;
 }
);

  const worked=new Set(F.shifts.map(s=>s.id_tienda+"|"+s.personId+"|"+s.day_int));

  const months=uniq(dates.map(d=>{const {y,m}
=D.dayYMD(d);
return y+"-"+m;
}
));

  const monthly=[],detail=[];

  employees.forEach(emp=>{
    months.forEach(ym=>{ const [Y,M]=ym.split("-").map(Number);

      const md=dates.filter(d=>{const {y,m}
=D.dayYMD(d);
return y===Y&&m===M;
}
);

      const sats=md.filter(d=>D.weekdayISO(d)===5), suns=md.filter(d=>D.weekdayISO(d)===6);

      const pairs=sats.filter(s=>dateSet.has(s+1)).map(s=>[s,s+1]);

      const freeSat=sats.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      const freeSun=suns.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      let freeWk=0;

      pairs.forEach(([sat,sun])=>{ const sf=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sat), su=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sun);

        if(sf&&su)freeWk++;

        detail.push({id_tienda:emp.id_tienda,personId:emp.personId,ini:sat,fin:`${ddmm(sat)}
 - ${ddmm(sun)}
`,sat:sf,sun:su,libre:sf&&su}
);
 }
);

      monthly.push({id_tienda:emp.id_tienda,personId:emp.personId,app:emp.app,Mes:`${Y}
-${String(M).padStart(2,"0")}
`,
        fs_eval:pairs.length,fs_libres:freeWk,sab_eval:sats.length,sab_libres:freeSat,dom_eval:suns.length,dom_libres:freeSun}
);
 }
);

  }
);

  return {monthly,detail}
;

}

/* ---------- Fines de semana ---------- */
function prepWeekend(F){
  const dates=[...F.dataSet].sort((a,b)=>a-b);
 const dateSet=F.dataSet;

  // empleados: applicable máximo (o null si algún null)
  const g=groupBy(F.weekly,w=>w.id_tienda+"|"+w.personId);

  const employees=[...g.values()].map(rs=>{ const vals=rs.map(r=>r.applicableWorkingHours);

    const app = vals.some(v=>v==null)?null:Math.max(...vals.map(Number));

    return {id_tienda:rs[0].id_tienda,personId:rs[0].personId,app}
;
 }
);

  const worked=new Set(F.shifts.map(s=>s.id_tienda+"|"+s.personId+"|"+s.day_int));

  const months=uniq(dates.map(d=>{const {y,m}
=D.dayYMD(d);
return y+"-"+m;
}
));

  const monthly=[],detail=[];

  employees.forEach(emp=>{
    months.forEach(ym=>{ const [Y,M]=ym.split("-").map(Number);

      const md=dates.filter(d=>{const {y,m}
=D.dayYMD(d);
return y===Y&&m===M;
}
);

      const sats=md.filter(d=>D.weekdayISO(d)===5), suns=md.filter(d=>D.weekdayISO(d)===6);

      const pairs=sats.filter(s=>dateSet.has(s+1)).map(s=>[s,s+1]);

      const freeSat=sats.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      const freeSun=suns.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      let freeWk=0;

      pairs.forEach(([sat,sun])=>{ const sf=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sat), su=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sun);

        if(sf&&su)freeWk++;

        detail.push({id_tienda:emp.id_tienda,personId:emp.personId,ini:sat,fin:`${ddmm(sat)}
 - ${ddmm(sun)}
`,sat:sf,sun:su,libre:sf&&su}
);
 }
);

      monthly.push({id_tienda:emp.id_tienda,personId:emp.personId,app:emp.app,Mes:`${Y}
-${String(M).padStart(2,"0")}
`,
        fs_eval:pairs.length,fs_libres:freeWk,sab_eval:sats.length,sab_libres:freeSat,dom_eval:suns.length,dom_libres:freeSun}
);
 }
);

  }
);

  return {monthly,detail}
;

}

function fmt(v, d=0){

}

  const m=/[T ](\d{2}

}

  if(start>afternoon) return "TARDE";

  return "CENTRAL";

}

function prepWeekend(F){
  const dates=[...F.dataSet].sort((a,b)=>a-b);
 const dateSet=F.dataSet;

  // empleados: applicable máximo (o null si algún null)
  const g=groupBy(F.weekly,w=>w.id_tienda+"|"+w.personId);

  const employees=[...g.values()].map(rs=>{ const vals=rs.map(r=>r.applicableWorkingHours);

    const app = vals.some(v=>v==null)?null:Math.max(...vals.map(Number));

    return {id_tienda:rs[0].id_tienda,personId:rs[0].personId,app}
;
 }
);

  const worked=new Set(F.shifts.map(s=>s.id_tienda+"|"+s.personId+"|"+s.day_int));

  const months=uniq(dates.map(d=>{const {y,m}
=D.dayYMD(d);
return y+"-"+m;
}
));

  const monthly=[],detail=[];

  employees.forEach(emp=>{
    months.forEach(ym=>{ const [Y,M]=ym.split("-").map(Number);

      const md=dates.filter(d=>{const {y,m}
=D.dayYMD(d);
return y===Y&&m===M;
}
);

      const sats=md.filter(d=>D.weekdayISO(d)===5), suns=md.filter(d=>D.weekdayISO(d)===6);

      const pairs=sats.filter(s=>dateSet.has(s+1)).map(s=>[s,s+1]);

      const freeSat=sats.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      const freeSun=suns.filter(s=>!worked.has(emp.id_tienda+"|"+emp.personId+"|"+s)).length;

      let freeWk=0;

      pairs.forEach(([sat,sun])=>{ const sf=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sat), su=!worked.has(emp.id_tienda+"|"+emp.personId+"|"+sun);

        if(sf&&su)freeWk++;

        detail.push({id_tienda:emp.id_tienda,personId:emp.personId,ini:sat,fin:`${ddmm(sat)}
 - ${ddmm(sun)}
`,sat:sf,sun:su,libre:sf&&su}
);
 }
);

      monthly.push({id_tienda:emp.id_tienda,personId:emp.personId,app:emp.app,Mes:`${Y}
-${String(M).padStart(2,"0")}
`,
        fs_eval:pairs.length,fs_libres:freeWk,sab_eval:sats.length,sab_libres:freeSat,dom_eval:suns.length,dom_libres:freeSun}
);
 }
);

  }
);

  return {monthly,detail}
;

}

  function weekendCounts(year, month, workedSet) {
    const daysInMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();

    let freeSat = 0, freeSun = 0, complete = 0;

    for (let d = 1;
 d <= daysInMonth;
 d++) {
      const day = dayFromYMD(year, month, d);

      const wd = weekdayISO(day);

      if (wd === 5) { // sábado
        if (!workedSet.has(day)) freeSat++;

        const nxt = day + 1;

        if (!workedSet.has(day) && dayYMD(nxt).m === month && !workedSet.has(nxt)) complete++;

      }

        if (!workedSet.has(day)) freeSun++;

      }

    }

    return [complete, freeSat, freeSun];

  function weekendCounts(year, month, workedSet) {
    const daysInMonth = new Date(Date.UTC(year, month, 0)).getUTCDate();

    let freeSat = 0, freeSun = 0, complete = 0;

    for (let d = 1;
 d <= daysInMonth;
 d++) {
      const day = dayFromYMD(year, month, d);

      const wd = weekdayISO(day);

      if (wd === 5) { // sábado
        if (!workedSet.has(day)) freeSat++;

        const nxt = day + 1;

        if (!workedSet.has(day) && dayYMD(nxt).m === month && !workedSet.has(nxt)) complete++;

      }
 else if (wd === 6) { // domingo
        if (!workedSet.has(day)) freeSun++;

      }

    }

    return [complete, freeSat, freeSun];

  }

  function analyzeWeeklyHours(shifts, employeeMonths, employeeMonthsMeta, dataDates, absences, presence, presenceMeta) {
    const tol = CONFIG.calculation.weekly_hours_tolerance;

    const hoursByWeek = new Map();
       // store|person|monday -> hours
    const hoursByEmp = new Map();
        // store|person -> [hours]
    const workedDays = new Map();
        // store|person -> Set(day)
    const applicableByEmp = new Map();
   // store|person -> applicable
    const applicableMeta = new Map();

    const absencesByDay = new Map();
     // store|person|day -> Set(type)

    for (const k of employeeMonths.keys()) {
      const meta = employeeMonthsMeta.get(k);

      const emp = meta.store + "\u0001" + meta.person;

      applicableByEmp.set(emp, employeeMonths.get(k));

      applicableMeta.set(emp, { store: meta.store, person: meta.person }
);

    }

    for (const s of shifts) {
      const emp = s.store + "\u0001" + s.person;

      const wk = emp + "\u0001" + weekStart(s.workDay);

      hoursByWeek.set(wk, (hoursByWeek.get(wk) || 0) + s.workedHours);

      if (!hoursByEmp.has(emp)) hoursByEmp.set(emp, []);

      hoursByEmp.get(emp).push(s.workedHours);

      if (!workedDays.has(emp)) workedDays.set(emp, new Set());

      workedDays.get(emp).add(s.workDay);

      if (s.applicable !== null && s.applicable !== undefined && s.applicable !== "") applicableByEmp.set(emp, s.applicable);

    }

    for (const a of absences) {
      const k = a.store + "\u0001" + a.person + "\u0001" + a.day;

      if (!absencesByDay.has(k)) absencesByDay.set(k, new Set());

      absencesByDay.get(k).add(a.type);

    }

    if (!dataDates.size) return [];

    const allDates = [...dataDates];

    const firstWeek = weekStart(Math.min(...allDates));

    const lastWeek = weekStart(Math.max(...allDates));

    const weekStarts = [];

    for (let d = firstWeek;
 d <= lastWeek;
 d++) if (weekdayISO(d) === 0) weekStarts.push(d);

      String(a.meta.person) < String(b.meta.person) ? -1 : String(a.meta.person) > String(b.meta.person) ? 1 : 0);


    const rows = [];

    for (const item of empList) {
      const emp = item.emp, store = item.meta.store, person = item.meta.person, applicable = item.applicable;

      const values = hoursByEmp.get(emp) || [];

      const averageDaily = values.length ? pyround(values.reduce((a, b) => a + b, 0) / values.length, 4) : null;

      const presenceDates = presence.get(emp) || new Set();

      const allAbsenceDates = new Set();

      for (const [k, types] of absencesByDay) {
        const parts = k.split("\u0001");

        if (parts[0] == store && parts[1] == person && types.size) allAbsenceDates.add(Number(parts[2]));

      }

      const wd = workedDays.get(emp);

      let subset = presenceDates.size > 0;

      for (const d of presenceDates) if (!allAbsenceDates.has(d)) { subset = false;
 break;

 for (let i = 0;
 i < 7;
 i++) weekDays.push(monday + i);
