function runValidation(data, scheduleSource) {

    const incidents = runRules(ex.shifts);

    const summaries = buildMonthlySummaries(ex.shifts, ex.employeeMonths, ex.employeeMonthsMeta, incidents);

    const dataDates = collectDataDates(data);

    const weekly = analyzeWeeklyHours(ex.shifts, ex.employeeMonths, ex.employeeMonthsMeta, dataDates, ex.absences, ex.presence, ex.presenceMeta);

    const balance = analyzeShiftBalance(ex.shifts);

    const absenceDaily = analyzeDailyAbsences(ex.absences, dataDates);


    const shiftsFrame = ex.shifts.map(s => ({
      id_tienda: s.store, personId: s.person, applicableWorkingHours: s.applicable,
      day: dayISO(s.workDay), day_int: s.workDay,
      hora_inicio: new Date(s.shiftStart).toISOString().slice(0, 19), hora_fin: new Date(s.shiftEnd).toISOString().slice(0, 19),
      horas_totales: s.workedHours, duracion_descanso: s.breakHours, franja_turno: classifyShiftPeriod(s.shiftStart),
    }
));

    const incidentsFrame = incidents.map(i => ({
      id_tienda: i.store, personId: i.person, mes: i.month, tipo_incidencia: i.type,
      fecha_inicio: dayISO(i.start), fecha_inicio_int: i.start, fecha_fin: dayISO(i.end), fecha_fin_int: i.end,
      valor_observado: i.observed, limite: i.limit, detalle: i.detail,
    }
));

    const absencesFrame = ex.absences.map(a => ({
      id_tienda: a.store, personId: a.person, fecha: dayISO(a.day), fecha_int: a.day, tipo_ausencia: a.type, estado: a.status,
    }
));

    return {
      schedule_source: scheduleSource, store_id: ex.storeId,
      data_dates: [...dataDates].sort((a, b) => a - b),
      shifts: shiftsFrame, summaries, incidents: incidentsFrame, weekly,
      absences: absencesFrame, shift_balance: balance, absence_daily: absenceDaily,
    }
